306 size_t _recvRespDescr(lidarRespDescr& resp, uint32_t startFlagTimeout = 5000) { // read serial data untill both RESP_DESCR_START_FLAGS are received. startFlagTimeout is in micros, returns the number of bytes read
307 lidarSerial.setTimeout(slowSerialTimout);
308 uint32_t startTime = micros();
309 uint8_t syncProgress = 0;
310 while (((micros() - startTime) < startFlagTimeout) && (syncProgress < sizeof(RESP_DESCR_START_FLAGS))) {
311 if (lidarSerial.available()) {
312 uint8_t potentialSyncByte = lidarSerial.read();
313 if (potentialSyncByte == RESP_DESCR_START_FLAGS[syncProgress]) {
314 syncProgress++;
315 }
316 }
317 }
318 if (syncProgress < sizeof(RESP_DESCR_START_FLAGS)) {
319#ifdef lidarDebugSerial
320 lidarDebugSerial.print("rplidar: _recvRespDescr() "); lidarDebugSerial.print(syncProgress); lidarDebugSerial.print(" < "); lidarDebugSerial.println(sizeof(RESP_DESCR_START_FLAGS));
321#endif
322 return(0); //sync bytes were not received (within startFlagTimeout)
323 } // (else) //sync bytes were all received, time to read the message
324 return(lidarSerial.readBytes(&resp, sizeof(resp)));
325 }

433 uint16_t _varbitscale_decode(uint16_t major, uint8_t& scaleLevel) { // taken from SDK (and modified slightly)
434 static const int32_t VBS_SCALED_BASE[] = {
435 3328, // SL_LIDAR_VARBITSCALE_X16_DEST_VAL // thijs: 3328 = 1101 0000 0000 = (13)<<8 ?
436 1792, // SL_LIDAR_VARBITSCALE_X8_DEST_VAL // thijs: 1792 = 0111 0000 0000 = (7)<<8 ?
437 1280, // SL_LIDAR_VARBITSCALE_X4_DEST_VAL // thijs: 1280 = 0101 0000 0000 = (5)<<8 ?
438 512, // SL_LIDAR_VARBITSCALE_X2_DEST_VAL // thijs: 512 = 0010 0000 0000 = (2)<<8 ?
439 0 };
440 static const uint16_t VBS_TARGET_BASE[] = {
441 (1 << 14), // SL_LIDAR_VARBITSCALE_X16_SRC_BIT
442 (1 << 12), // SL_LIDAR_VARBITSCALE_X8_SRC_BIT
443 (1 << 11), // SL_LIDAR_VARBITSCALE_X4_SRC_BIT
444 (1 << 9), // SL_LIDAR_VARBITSCALE_X2_SRC_BIT
445 0 };
446 for (size_t i = 0; i < 5; ++i) {
447 int32_t remain = ((int32_t)major - VBS_SCALED_BASE[i]);
448 if (remain >= 0) {
449 scaleLevel = 5 - 1 - i;
450 return(VBS_TARGET_BASE[i] + (remain << scaleLevel));
451 }
452 }
453 return(0);
454 }
